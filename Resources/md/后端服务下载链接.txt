Docker 一键启动配套服务端（和 Nacos2.3.0 版本对齐）
1. Nacos 2.3.0 单机
bash
运行
docker run -d --name nacos230 -p 8848:8848 -e MODE=standalone nacos/nacos-server:v2.3.0
2. Sentinel Dashboard 1.8.6
bash
运行
docker run -d --name sentinel -p 8080:8080 bladex/sentinel-dashboard:1.8.6
3. Seata 1.7.1
bash
运行
docker run -d --name seata -p 8091:8091 seataio/seata-server:1.7.1
4. RocketMQ 5.1.4
bash
运行
# NameServer
docker run -d --name rmqnamesrv -p 9876:9876 apache/rocketmq:5.1.4 sh mqnamesrv
# Broker
docker run -d --name rmqbroker --link rmqnamesrv:namesrv -e "NAMESRV_ADDR=namesrv:9876" -p 10909:10909 apache/rocketmq:5.1.4 sh mqbroker
